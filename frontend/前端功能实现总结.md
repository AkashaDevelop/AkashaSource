func GetSystemMonitor(c *gin.Context) {

### 路由注册

在 `backend/router/api.go` 中添加：

```go
// 管理员路由
admin := router.Group("/api/admin")
admin.Use(middleware.AdminAuth())
{
    // 系统监控
    admin.GET("/system/monitor", controller.GetSystemMonitor)

    // 渠道亲和性
    admin.GET("/channel-affinity/rules", controller.GetAffinityRules)
    admin.POST("/channel-affinity/rules", controller.SaveAffinityRules)
    admin.GET("/channel-affinity/stats", controller.GetAffinityStats)
    admin.POST("/channel-affinity/cache/clear", controller.ClearAffinityCache)

    // 批量渠道操作
    admin.POST("/channels/batch-status", controller.BatchUpdateChannelStatus)
    admin.POST("/channels/batch-priority", controller.BatchUpdateChannelPriority)
    admin.POST("/channels/batch-delete", controller.BatchDeleteChannels)
}

// 用户路由
user := router.Group("/api/user")
user.Use(middleware.UserAuth())
{
    user.GET("/notification-settings", controller.GetNotificationSettings)
    user.POST("/notification-settings", controller.SaveNotificationSettings)
    user.POST("/notification-test", controller.TestNotification)
}
```
